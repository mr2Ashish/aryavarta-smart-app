function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Inbound_Leads") || ss.insertSheet("Inbound_Leads");
  var d = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(), d.name, d.company, d.location, d.phone, d.email,
    d.industry, d.panelType, d.ratingKw + " kW", d.make, d.estimatedBudget
  ]);
  return ContentService.createTextOutput(JSON.stringify({status:"ok"}))
    .setMimeType(ContentService.MimeType.JSON);
}
